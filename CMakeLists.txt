cmake_minimum_required(VERSION 3.20)
project(RestoreChineseSymbol LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(symbol_schemes STATIC src/scheme_config.cpp src/scheme_manager.cpp)
target_include_directories(symbol_schemes PUBLIC src)
if(MSVC)
    target_compile_options(symbol_schemes PRIVATE /W4 /utf-8)
else()
    target_compile_options(symbol_schemes PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
    add_executable(RestoreChineseSymbol WIN32 src/main.cpp)
    target_compile_definitions(RestoreChineseSymbol PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
    target_link_libraries(RestoreChineseSymbol PRIVATE symbol_schemes user32 shell32)
    add_custom_command(TARGET RestoreChineseSymbol POST_BUILD
        COMMAND ${CMAKE_COMMAND}
            "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}/examples/schemes.ini"
            "-DDESTINATION=$<TARGET_FILE_DIR:RestoreChineseSymbol>/schemes.ini"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/CopyDefaultConfig.cmake"
        VERBATIM)
    if(MINGW)
        target_link_options(RestoreChineseSymbol PRIVATE -municode -static)
    endif()
    if(MSVC)
        target_compile_options(RestoreChineseSymbol PRIVATE /W4 /utf-8)
    else()
        target_compile_options(RestoreChineseSymbol PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(remapper_test tests/remapper_test.cpp)
    target_include_directories(remapper_test PRIVATE src)
    if(MINGW)
        target_link_options(remapper_test PRIVATE -static)
    endif()
    add_test(NAME key_sequences COMMAND remapper_test)
    add_executable(scheme_config_test tests/scheme_config_test.cpp)
    target_link_libraries(scheme_config_test PRIVATE symbol_schemes)
    if(MINGW)
        target_link_options(scheme_config_test PRIVATE -static)
    endif()
    foreach(test_target remapper_test scheme_config_test)
        if(MSVC)
            target_compile_options(${test_target} PRIVATE /W4 /utf-8)
        else()
            target_compile_options(${test_target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endforeach()
    add_test(NAME scheme_configuration COMMAND scheme_config_test "${CMAKE_CURRENT_SOURCE_DIR}/examples/schemes.ini")
endif()
